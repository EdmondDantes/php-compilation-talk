Build time Специализация

Известная конфигурация удаляет недоступную ветвь

// исходная программа
if (BUILD_GATEWAY === 'new') {
    $gateway = new NewGateway();
} else {
    $gateway = new LegacyGateway();
}
$gateway->charge($order);
// сборка BUILD_GATEWAY=new
$gateway = new NewGateway();
$gateway->charge($order);

// LegacyGateway недостижим
// точный класс известен оптимизатору

// условие: значение фиксируется при сборке, а не читается в runtime
